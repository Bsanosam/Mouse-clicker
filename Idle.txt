Shinies
	*GoldMouse
		on click:log Woop
		movement:onBottom moveTop fade bounce:0.05
		frequency:60
		frequency variation:30
		icon:stuff/luckyBunny.png
		class:bigButton
		on click:
			$boost=1
			if (have clover) $boost=4
			if (chance(25%))
				//get at least 7, or between 1 and 3 minutes of our mouse production
				$amount=max(7,random(mice:ps*60*1,mice:ps*60*3))*$boost
				toast The gold mouse grants you [$amount] mice.
				yield $amount mice
			else
				$amount=1*$boost
				toast The Gold mouse grants you [$amount] golden cheese[s?$amount]!
				yield $amount goldenCheese
			end
		end
